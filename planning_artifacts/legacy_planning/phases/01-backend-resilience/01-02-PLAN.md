```python
# ─── Idempotency — dedup double-tap uploads ─────────────────────────────────
_DEDUP_TTL_SECONDS = 300  # 5 minutes
_dedup_cache: dict[str, tuple[str, float]] = {}  # hash -> (job_id, timestamp)


def _dedup_key(filename: str, file_size: int, user_id: int | None) -> str:
    """SHA256 of (filename + file_size + user_id) for dedup."""
    raw = f"{filename}:{file_size}:{user_id}"
    return hashlib.sha256(raw.encode()).hexdigest()


def _dedup_cleanup() -> None:
    """Remove expired entries from dedup cache."""
    now = time.time()
    expired = [k for k, (_, ts) in _dedup_cache.items() if now - ts > _DEDUP_TTL_SECONDS]
    for k in expired:
        del _dedup_cache[k]
```

**Change 3 — Add dedup check inside `process_invoice`, after `file_bytes = b"".join(chunks)` and before `job_id = str(uuid.uuid4())`:**

Insert this block between the file reading and the job creation:

```python
    # ── Idempotency check (dedup double-tap) ──────────────────────
    _dedup_cleanup()
    dedup_hash = _dedup_key(filename, len(file_bytes), user_id)
    existing = _dedup_cache.get(dedup_hash)
    if existing:
        existing_job_id, _ = existing
        logger.info("Dedup hit job=%s file=%s user=%s — returning existing job", existing_job_id, filename, user_id)
        return JSONResponse(
```
